% Macro to insert the folding peptide image based on page progress
\newcommand{\insertpeptide}{%
  \ifnum\value{page}>0 % Ensure page number is positive
    \setrefcountdefault{240}% Set fallback to 240 if LastPage is not yet defined
    \edef\totalpagesnum{\getpagerefnumber{LastPage}}%
    % Calculate index: round(1 + (page - 1)/(total_pages - 1) * 239) and clamp between 1 and 240
    \edef\imgindex{\fpeval{min(240, max(1, round(1 + (\thepage - 1) / (\totalpagesnum - 1) * 239, 0)))}}%
    % Load the image (scaled to 2.1cm height so it is clearly visible in the footer)
    \raisebox{-0.32\height}{\includegraphics[height=2.1cm]{FOLDING_PEPTIDE_PICTURE/folding_\imgindex.png}}%
  \fi
}

% Define footers (page numbers and folding peptide on outer corners)
